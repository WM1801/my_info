Если тебе нужны полные исходники для поиска функций:
[source, bash]
----
# Ubuntu/Debian:
sudo apt update
sudo apt install linux-source

# Проверь, куда установилось:
ls /usr/src/linux-source-*

# Распакуй (если нужно):
cd /usr/src
sudo tar -xf linux-source-*.tar.xz

# Теперь ищи:
grep -r "get_random_u32" /usr/src/linux-source-*/include/linux/random.h
----
