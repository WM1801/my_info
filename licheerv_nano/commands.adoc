== Команды дающие понимание о системе
----
# 1. Проверяем права доступа к /dev/mem
ls -l /dev/mem

# 2. Проверяем, включен ли STRICT_DEVMEM
cat /proc/config.gz 2>/dev/null | gunzip | grep STRICT_DEVMEM
# или
dmesg | grep -i "strict_devmem"

# 3. Пробуем прочитать "запрещенный" адрес (ядро)
devmem 0x80000000

# 4. Пробуем прочитать несуществующий адрес
devmem 0xFFFFFFFF

# 5. Смотрим, какие регионы памяти доступны
cat /proc/iomem | head -50

# 6. Команда для просмотра системных сообщений ядра Linux в реальном времени
tail -f /var/log/kern.log 

----
